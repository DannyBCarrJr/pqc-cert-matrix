== chain: ecdsa (root=EC int=EC leaf=EC) ==
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/ecdsa/root.key
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/ecdsa/int.key
$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out chains/ecdsa/leaf.key
$ openssl req -x509 -new -key chains/ecdsa/root.key -out chains/ecdsa/root.crt -subj /CN=PQM Matrix ecdsa Root -days 90 -addext basicConstraints=critical,CA:TRUE -addext keyUsage=critical,keyCertSign,cRLSign
$ openssl req -new -key chains/ecdsa/int.key -out chains/ecdsa/int.csr -subj /CN=PQM Matrix ecdsa Intermediate
$ openssl x509 -req -in chains/ecdsa/int.csr -CA chains/ecdsa/root.crt -CAkey chains/ecdsa/root.key -out chains/ecdsa/int.crt -days 60 -extfile /dev/fd/63
Certificate request self-signature ok
subject=CN=PQM Matrix ecdsa Intermediate
$ openssl req -new -key chains/ecdsa/leaf.key -out chains/ecdsa/leaf.csr -subj /CN=matrix.test
$ openssl x509 -req -in chains/ecdsa/leaf.csr -CA chains/ecdsa/int.crt -CAkey chains/ecdsa/int.key -out chains/ecdsa/leaf.crt -days 30 -extfile /dev/fd/63
Certificate request self-signature ok
subject=CN=matrix.test
-- verify --
$ openssl verify -CAfile chains/ecdsa/root.crt -untrusted chains/ecdsa/int.crt chains/ecdsa/leaf.crt
chains/ecdsa/leaf.crt: OK
-- DER sizes (bytes) --
ecdsa/root: 392
ecdsa/int: 437
ecdsa/leaf: 464
